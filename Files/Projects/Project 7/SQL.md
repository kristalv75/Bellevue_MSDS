## Table Creation

CREATE TABLE CUST_PAYMENT (
CUST_PAY_ID INT NOT NULL UNIQUE,
CUST_ID INT NOT NULL,
CUST_PAY_AMT NUMERIC(8,2) NOT NULL,
CUST_PAY_DATE DATE NOT NULL,
CUST_PAY_REMAINING NUMERIC(8,2) NOT NULL,
PRIMARY KEY (CUST_PAY_ID))

CREATE TABLE SUP_PAYMENT (
SUP_PAY_ID INT NOT NULL UNIQUE,
SUP_ID INT NOT NULL,
SUP_PAY_AMT NUMERIC(8,2) NOT NULL,
SUP_PAY_DATE DATE NOT NULL,
SUP_PAY_REMAINING NUMERIC(8,2) NOT NULL,
PRIMARY KEY (SUP_PAY_ID))

CREATE TABLE CUSTOMER (
CUST_ID INT NOT NULL UNIQUE,
CUST_FIRST_NAME VARCHAR(30) NOT NULL,
CUST_LAST_NAME VARCHAR(30) NOT NULL,
CUST_ADD1 VARCHAR(30) NOT NULL,
CUST_ADD2 VARCHAR(30) ,
CUST_CITY VARCHAR(30) NOT NULL,
CUST_STATE VARCHAR(30) NOT NULL,
CUST_ZIP VARCHAR(10) NOT NULL,
CUST_PHONE VARCHAR(15) NOT NULL,
CUST_EMAIL VARCHAR(30) NOT NULL,
PRIMARY KEY (CUST_ID))

CREATE TABLE SUPPLIER (
SUP_ID INT NOT NULL UNIQUE,
SUP_COMPANY_NAME VARCHAR(30) NOT NULL,
SUP_ADD1 VARCHAR(30) NOT NULL,
SUP_ADD2 VARCHAR(30) ,
SUP_CITY VARCHAR(30) NOT NULL,
SUP_STATE VARCHAR(30) NOT NULL,
SUP_ZIP VARCHAR(10) NOT NULL,
SUP_PHONE VARCHAR(15) NOT NULL,
SUP_EMAIL VARCHAR(30) NOT NULL,
PRIMARY KEY (SUP_ID))

CREATE TABLE BID (
BID_ID INT NOT NULL UNIQUE,
CUST_ID INT NOT NULL,
BID_DESC VARCHAR(50) NOT NULL,
BID_DATE DATE NOT NULL,
BID_MAT_PRICE NUMERIC(8,2) NOT NULL,
BID_LABOR_PRICE NUMERIC(8,2) NOT NULL,
BID_TOT_PRICE NUMERIC(8,2) NOT NULL,
BID_LABOR_HOURS NUMERIC(8,2) NOT NULL,
PRIMARY KEY (BID_ID),
FOREIGN KEY (CUST_ID) REFERENCES CUSTOMER)

CREATE TABLE REPAIR (
REPAIR_ID INT NOT NULL UNIQUE,
REPAIR_DESC VARCHAR(50) NOT NULL,
BID_ID INT NOT NULL,
REPAIR_MAT_PRICE NUMERIC(8,2) NOT NULL,
REPAIR_LABOR_PRICE NUMERIC(8,2) NOT NULL,
REPAIR_TOT_PRICE NUMERIC(8,2) NOT NULL,
PRIMARY KEY (REPAIR_ID),
FOREIGN KEY (BID_ID) REFERENCES BID)

CREATE TABLE MATERIAL (
REPAIR_ID INT NOT NULL,
SUP_ID INT NOT NULL,
MAT_ITEM VARCHAR(50)  NOT NULL,
MAT_QUANT INT NOT NULL,
MAT_UNIT_PRICE NUMERIC(8,2) NOT NULL,
MAT_TOT_PRICE NUMERIC(8,2) NOT NULL,
PRIMARY KEY (REPAIR_ID, SUP_ID),
FOREIGN KEY (REPAIR_ID) REFERENCES REPAIR,
FOREIGN KEY (SUP_ID) REFERENCES SUPPLIER)
